    /**
     * FOUND-04 — MDX content collections must NOT bundle into the Worker bundle.
     *
     * The Worker bundle is whatever Astro emits as server-side JS (under dist/_worker.js

     *
     * Pick a known sentence fragment from a real project MDX file (e.g., the
     * literal text "Problem:" or a project-specific phrase) and assert ZERO
     * matches across all dist/**/*.js files emitted by the Worker bundle.
     */
    import { describe, it, expect, beforeAll } from "vitest";
    import { readFileSync, readdirSync, statSync, existsSync } from "node:fs";
    import { join } from "node:path";

    function listJsFiles(dir: string, acc: string[] = []): string[] {
      if (!existsSync(dir)) return acc;
      for (const entry of readdirSync(dir)) {
        const full = join(dir, entry);
        const st = statSync(full);
        if (st.isDirectory()) listJsFiles(full, acc);
        else if (entry.endsWith(".js") || entry.endsWith(".mjs")) acc.push(full);
      }
      return acc;
    }

      // that would NOT plausibly appear in framework or runtime code.

      let workerJsFiles: string[] = [];

        // and exclude dist/client/ (which is the static-asset tree, intentionally
        // contains compiled MDX HTML — that is correct).
        const distRoot = join(process.cwd(), "dist");
        const all = listJsFiles(distRoot);
        workerJsFiles = all.filter((p) => !p.includes("dist/client") && !p.includes("dist\client"));
